		t2.userid,

		count(1) login_size
	from

  PRIMARY KEY (`id`)
)

  `id` int(11) NOT NULL,
  `code` varchar(255) CHARACTER SET utf8mb4 COLLATE utf8mb4_bin DEFAULT NULL,
  `amount` varchar(255) CHARACTER SET utf8mb4 COLLATE utf8mb4_bin DEFAULT NULL,

  PRIMARY KEY (`id`) USING BTREE
)
sql
SELECT
	* 
FROM
	tx_draft t2 
WHERE
	t2.due_date > CURRENT_DATE ( ) 
	AND t2.due_date <= ( --  查询 从当前日期的后一天开始 计算 第三天的日期
	SELECT date FROM tx_holidays t WHERE t.flag = 0 AND t.date > CURRENT_DATE ( ) --  只要 下面的2数之和  为3  则是三个工作日
	
ORDER BY

join t_class c

要求通过 sql 实现 1条记录 假设 attr 只有 b1 b2 b3这三个值

sql
SELECT 
    t20.id,
    t20.a1,
    t20.a2,
    t20.a3,
    MAX(CASE WHEN t20_1.attr = 'b1' THEN t20_1.attr_val END) AS b1,
    MAX(CASE WHEN t20_1.attr = 'b2' THEN t20_1.attr_val END) AS b2,
    MAX(CASE WHEN t20_1.attr = 'b3' THEN t20_1.attr_val END) AS b3
FROM t20
LEFT JOIN t20_1 ON t20.id = t20_1.id
GROUP BY t20.id;
-- 若某个 attr（如 b1）不存在，对应的列值为 NULL。
-- 若存在多个相同 id 和 attr 的组合，MAX 会选取最大的值（需根据实际需求调整聚合函数

select name,
 sum(IF(t.mn=1,t.total_count,0)) '1季度销售量', 
 sum(IF(t.mn=2,t.total_count,0)) '2季度销售量', 
 sum(IF(t.mn=3,t.total_count,0)) '3季度销售量', 
 sum(IF(t.mn=4,t.total_count,0)) '4季度销售量'
  from(
 select name, total_count, 
  case when MONTH(create_date) in(1,2,3)   then 1
   when MONTH(create_date)in(4,5,6) then 2
   when MONTH(create_date)in(7,8,9) then 3    
   when MONTH(create_date)in(10,11,12) then 4
   end as mn
  from tb_sales
)   t
  group by t.name

灰度发布 ​

nginx
# ============================================================
# 方案一：基于权重的灰度（最常用）
# 按比例将流量分配到新旧版本
# ============================================================
upstream backend {
    server 127.0.0.1:8080 weight=90;  # 旧版本，90% 流量
    server 127.0.0.1:8086 weight=10;  # 新版本，10% 流量
}

server {
    listen 80;


    location / {
        proxy_pass http://backend;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
    }
}

# ============================================================
# 方案二：基于请求头的灰度
# 请求头带 X-Gray: true 的用户访问新版本
# ============================================================
upstream prod {
    server 127.0.0.1:8080;
}

upstream gray {
    server 127.0.0.1:8086;
}

server {
    listen 80;


    # 默认走旧版本
    set $backend prod;

    # 匹配灰度条件
    if ($http_x_gray = "true") {
        set $backend gray;
    }

    location / {
        proxy_pass http://$backend;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
    }
}

# ============================================================
# 方案三：基于 Cookie 的灰度
# 带 cookie gray=1 的用户访问新版本
# ============================================================
server {
    listen 80;


    set $backend prod;

    if ($cookie_gray = "1") {
        set $backend gray;
    }

    location / {
        proxy_pass http://$backend;
        proxy_set_header Host $host;
    }
}

# ============================================================
# 方案四：基于 IP 的灰度
# 指定 IP 访问新版本（内测用户）
# ============================================================
server {
    listen 80;


    set $backend prod;

    # 灰度 IP 白名单
    if ($remote_addr = "192.168.1.100") {
        set $backend gray;
    }
    if ($remote_addr = "192.168.1.101") {
        set $backend gray;
    }

    location / {
        proxy_pass http://$backend;
        proxy_set_header Host $host;
    }
}

    server 127.0.0.1:8080;
}

upstream green {
    server 127.0.0.1:8086;

# 当前活跃环境（切换时只改这一行）
# 改成 @green 即切到新版本
upstream active {
    server 127.0.0.1:8080;  # blue
    # server 127.0.0.1:8086;  # green

    listen 80;

流量复制 生产的流量导入到测试 ​

nginx
# ============================================================
# Nginx 流量镜像配置
# 功能：将线上流量复制一份到测试环境，不影响正式响应
# ============================================================

# 正式环境
upstream prod {
    server 127.0.0.1:8080;
}

# 测试环境（镜像目标）
upstream test {
    server 127.0.0.1:8086;
}

server {
    listen 80;


    # ----------------------------------------------------------
    # 全量镜像：所有请求都复制一份到测试环境
    # ----------------------------------------------------------
    location / {
        mirror /mirror;                  # 开启镜像，指定镜像接口
        mirror_request_body on;          # 镜像请求体（POST/PUT 等）

        proxy_pass http://prod;
        proxy_set_header Host              $host;
        proxy_set_header X-Real-IP         $remote_addr;
        proxy_set_header X-Forwarded-For   $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
    }

    # 镜像接收端（internal 表示外部无法直接访问）
    location /mirror {
        internal;
        proxy_pass http://test$request_uri;
        proxy_set_header Host              $host;
        proxy_set_header X-Real-IP         $remote_addr;
        proxy_set_header X-Forwarded-For   $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
    }

    # ----------------------------------------------------------
    # 可选：只镜像 GET 请求（写操作不镜像，避免数据污染）
